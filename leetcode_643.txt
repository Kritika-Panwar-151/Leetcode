class Solution {
    
    public double findMaxAverage(int[] nums, int k) 
    {
        //Sliding window
        double s=0.0;
        for(int i=0;i<k;i++)
        {
            s+=nums[i];
        }
        double avg=s/k;
        double a;
        for(int i=k;i<nums.length;i++)
        {
            s=s-nums[i-k]+nums[i];
            a=s/k;
            if(a>avg)
            {
                avg=a;
            }
        }
        return avg;

            {

            if(avg==0 || avg<a) 
            {

            } 
        } return avg;
        */
    }
    
}
